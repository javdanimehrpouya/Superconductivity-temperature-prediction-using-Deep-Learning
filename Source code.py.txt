import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from pathlib import Path
from typing import Tuple, Dict
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error
import tensorflow as tf
from tensorflow.keras.models import Model
from tensorflow.keras.layers import Input, Dense, Multiply
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import EarlyStopping, ModelCheckpoint
import joblib
import warnings

warnings.filterwarnings("ignore")

class SuperconductorAnalysis:
    def __init__(self, random_state: int = 42):
        """
        Initialize the SuperconductorAnalysis class.

        Args:
            random_state (int): Random seed for reproducibility
        """
        self.random_state = random_state
        self.model = None
        self.scaler = None
        self.feature_importance = None
        tf.random.set_seed(random_state)
        np.random.seed(random_state)

    def load_data(self, file_path: str) -> pd.DataFrame:
        """
        Load data from various file formats.

        Args:
            file_path (str): Path to the input file

        Returns:
            pd.DataFrame: Loaded dataframe
        """
        file_extension = Path(file_path).suffix.lower()

        try:
            if file_extension == '.csv':
                df = pd.read_csv(file_path)
            elif file_extension in ['.xlsx', '.xls']:
                df = pd.read_excel(file_path)
            elif file_extension == '.json':
                df = pd.read_json(file_path)
            else:
                raise ValueError(f"Unsupported file format: {file_extension}")

            print(f"Successfully loaded data with shape: {df.shape}")
            return df

        except Exception as e:
            print(f"Error loading file: {str(e)}")
            raise

    def preprocess_data(self, df: pd.DataFrame, target_column: str) -> Tuple[pd.DataFrame, pd.Series]:
        """
        Preprocess the data by handling missing values and selecting features.

        Args:
            df (pd.DataFrame): Input dataframe
            target_column (str): Name of the target column

        Returns:
            Tuple[pd.DataFrame, pd.Series]: Features and target
        """
        # Remove unnamed columns
        df = df.drop(columns=[col for col in df.columns if 'Unnamed' in col])

        # Select numerical features only
        features = df.select_dtypes(include=[np.number]).drop(columns=[target_column])
        target = df[target_column]

        # Handle missing values
        features = features.fillna(features.mean())

        # Print feature statistics
        print("\nFeature Statistics:")
        print(f"Number of features: {len(features.columns)}")
        print(f"Number of samples: {len(features)}")
        print(f"Missing values: {features.isnull().sum().sum()}")

        return features, target

    def build_attention_model(self, input_dim: int) -> Model:
        """
        Build an attention-based neural network model.

        Args:
            input_dim (int): Number of input features

        Returns:
            Model: Compiled Keras model
        """
        # Input layer
        inputs = Input(shape=(input_dim,))

        # First dense block
        x = Dense(256, activation='relu', kernel_initializer='he_normal')(inputs)
        x = Dense(128, activation='relu', kernel_initializer='he_normal')(x)
        x = Dense(64, activation='relu', kernel_initializer='he_normal')(x)

        # Attention mechanism
        attention_probs = Dense(input_dim, activation='softmax', name='attention_probs')(x)
        attention_mul = Multiply()([inputs, attention_probs])

        # Second dense block
        x = Dense(64, activation='relu', kernel_initializer='he_normal')(attention_mul)
        x = Dense(32, activation='relu', kernel_initializer='he_normal')(x)

        # Output layer
        output = Dense(1, activation='linear')(x)

        model = Model(inputs=[inputs], outputs=[output])
        model.compile(optimizer=Adam(learning_rate=0.001),
                      loss='mse',
                      metrics=['mae'])

        return model

    def train_model(self,
                    X: pd.DataFrame,
                    y: pd.Series,
                    validation_split: float = 0.2,
                    test_size: float = 0.1,
                    epochs: int = 200,
                    batch_size: int = 32) -> Dict:
        """
        Train the model and perform feature selection.

        Args:
            X (pd.DataFrame): Feature matrix
            y (pd.Series): Target variable
            validation_split (float): Validation split ratio
            test_size (float): Test set size
            epochs (int): Number of training epochs
            batch_size (int): Batch size for training

        Returns:
            Dict: Training history and metrics
        """
        # Split the data
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=self.random_state
        )

        # Scale the features
        self.scaler = StandardScaler()
        X_train_scaled = self.scaler.fit_transform(X_train)
        X_test_scaled = self.scaler.transform(X_test)

        # Build and train the model
        self.model = self.build_attention_model(X_train.shape[1])

        # Print model summary
        print("\nModel Architecture:")
        self.model.summary()

        # Callbacks
        early_stopping = EarlyStopping(
            monitor='val_loss',
            patience=20,
            restore_best_weights=True
        )

        checkpoint = ModelCheckpoint(
            'best_model.keras',
            monitor='val_loss',
            save_best_only=True
        )

        # Train the model
        print("\nTraining the model...")
        history = self.model.fit(
            X_train_scaled, y_train,
            validation_split=validation_split,
            epochs=epochs,
            batch_size=batch_size,
            callbacks=[early_stopping, checkpoint],
            verbose=1
        )

        # Get predictions
        y_pred = self.model.predict(X_test_scaled)

        # Calculate metrics
        metrics = {
            'r2_score': r2_score(y_test, y_pred),
            'mae': mean_absolute_error(y_test, y_pred),
            'rmse': np.sqrt(mean_squared_error(y_test, y_pred)),
            'history': history.history,
            'y_test': y_test,
            'y_pred': y_pred.flatten(),
            'X_test': X_test  # Save X_test for further analysis
        }

        # Print metrics
        print("\nModel Performance:")
        print(f"R² Score: {metrics['r2_score']:.4f}")
        print(f"MAE: {metrics['mae']:.4f}")
        print(f"RMSE: {metrics['rmse']:.4f}")

        # Extract feature importance
        attention_layer = self.model.get_layer('attention_probs')
        attention_model = Model(inputs=self.model.input, outputs=attention_layer.output)
        attention_weights = attention_model.predict(X_train_scaled)
        self.feature_importance = pd.DataFrame({
            'Feature': X.columns,
            'Importance': attention_weights.mean(axis=0)
        }).sort_values('Importance', ascending=False)

        return metrics

    def plot_training_history(self, history: Dict):
        """
        Plot training history.

        Args:
            history (Dict): Training history dictionary
        """
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 5))

        # Loss plot
        ax1.plot(history['loss'], label='Training Loss')
        ax1.plot(history['val_loss'], label='Validation Loss')
        ax1.set_title('Model Loss')
        ax1.set_xlabel('Epoch')
        ax1.set_ylabel('Loss')
        ax1.legend()
        ax1.grid(True)

        # MAE plot
        ax2.plot(history['mae'], label='Training MAE')
        ax2.plot(history['val_mae'], label='Validation MAE')
        ax2.set_title('Model MAE')
        ax2.set_xlabel('Epoch')
        ax2.set_ylabel('MAE')
        ax2.legend()
        ax2.grid(True)

        plt.tight_layout()
        plt.show()

    def plot_feature_importance(self, top_n: int = 20):
        """
        Plot feature importance.

        Args:
            top_n (int): Number of top features to display
        """
        plt.figure(figsize=(12, 8))
        sns.barplot(
            data=self.feature_importance.head(top_n),
            x='Importance',
            y='Feature',
            palette='viridis'
        )
        plt.title(f'Top {top_n} Most Important Features')
        plt.xlabel('Importance Score')
        plt.tight_layout()
        plt.show()

        # Print top features
        print(f"\nTop {top_n} Most Important Features:")
        print(self.feature_importance.head(top_n))

    def plot_predicted_vs_actual(self, y_test: pd.Series, y_pred: np.ndarray):
        """
        Plot comparison of predicted vs actual Tc values.

        Args:
            y_test (pd.Series): Actual Tc values
            y_pred (np.ndarray): Predicted Tc values
        """
        plt.figure(figsize=(10, 6))
        sns.scatterplot(x=y_test, y=y_pred, alpha=0.7)
        plt.plot([y_test.min(), y_test.max()], [y_test.min(), y_test.max()], 'r--')
        plt.xlabel('Actual Tc')
        plt.ylabel('Predicted Tc')
        plt.title('Comparison of Predicted vs Actual Tc')
        plt.grid(True)
        plt.tight_layout()
        plt.show()

    def plot_residuals(self, y_test: pd.Series, y_pred: np.ndarray):
        """
        Plot residuals of the predictions.

        Args:
            y_test (pd.Series): Actual Tc values
            y_pred (np.ndarray): Predicted Tc values
        """
        residuals = y_test - y_pred
        plt.figure(figsize=(10, 6))
        sns.histplot(residuals, kde=True, color='purple')
        plt.title('Distribution of Residuals')
        plt.xlabel('Residuals (Actual - Predicted)')
        plt.ylabel('Frequency')
        plt.tight_layout()
        plt.show()

    def plot_error_vs_actual(self, y_test: pd.Series, y_pred: np.ndarray):
        """
        Plot error magnitude vs actual Tc values.

        Args:
            y_test (pd.Series): Actual Tc values
            y_pred (np.ndarray): Predicted Tc values
        """
        errors = np.abs(y_test - y_pred)
        plt.figure(figsize=(10, 6))
        sns.scatterplot(x=y_test, y=errors, alpha=0.7, color='darkgreen')
        plt.xlabel('Actual Tc')
        plt.ylabel('Absolute Error')
        plt.title('Absolute Error vs Actual Tc')
        plt.grid(True)
        plt.tight_layout()
        plt.show()

    def save_model(self, path: str = 'superconductor_model'):
        """
        Save the model and scaler.

        Args:
            path (str): Path to save the model
        """
        if not os.path.exists(path):
            os.makedirs(path)

        self.model.save(f'{path}/model.keras')
        joblib.dump(self.scaler, f'{path}/scaler.pkl')
        self.feature_importance.to_csv(f'{path}/feature_importance.csv', index=False)

        print(f"Model and associated files saved to {path}")

    def load_model(self, path: str = 'superconductor_model'):
        """
        Load the saved model and scaler.

        Args:
            path (str): Path to load the model from
        """
        self.model = tf.keras.models.load_model(f'{path}/model.keras')
        self.scaler = joblib.load(f'{path}/scaler.pkl')
        self.feature_importance = pd.read_csv(f'{path}/feature_importance.csv')

        print("Model and associated files loaded successfully")

def main():
    """Main function to demonstrate usage"""
    # Initialize the analyzer
    analyzer = SuperconductorAnalysis()

    try:
        # Load data
        data_path = input("Enter the path to your data file: ")
        df = analyzer.load_data(data_path)

        # Get target column
        target_column = input("Enter the name of the target column (e.g., 'Tc'): ")

        # Preprocess data
        X, y = analyzer.preprocess_data(df, target_column)

        # Train model
        print("\nTraining model...")
        metrics = analyzer.train_model(X, y)

        # Plot results
        analyzer.plot_training_history(metrics['history'])
        analyzer.plot_feature_importance()
        analyzer.plot_predicted_vs_actual(metrics['y_test'], metrics['y_pred'])
        analyzer.plot_residuals(metrics['y_test'], metrics['y_pred'])
        analyzer.plot_error_vs_actual(metrics['y_test'], metrics['y_pred'])

        # Save model
        save_model = input("\nDo you want to save the model? (y/n): ")
        if save_model.lower() == 'y':
            path = input("Enter path to save model (default: 'superconductor_model'): ")
            path = path if path else 'superconductor_model'
            analyzer.save_model(path)

    except Exception as e:
        print(f"An error occurred: {str(e)}")

if __name__ == "__main__":
    main()
